# Model: BuildingState

Global game state. Updated every simulation tick. Never mutated directly — only via `applyDeltas()`.

## TypeScript Interface

```typescript
interface BuildingState {
  // Simulation clock
  tick: number;                      // Increments every simulation step
  gameTimeElapsed: number;           // Seconds since game start
  gameDurationTarget: number;        // Total game length in seconds (480–720)
  phase: GamePhase;

  // Building metrics — 0–100
  noiseLevel: number;                // Drives noise complaint events
  chaosLevel: number;                // Overall disorder — main lose meter
  stabilityIndex: number;            // Inverse of chaos — decays over time
  maintenanceScore: number;          // Building upkeep quality

  // Political state
  playerTrustAverage: number;        // Mean trustInPlayer across active residents
  supportCount: number;              // Residents with sentiment "supportive"
  hostileCount: number;              // Residents with sentiment "hostile"
  voteSentiment: number;             // 0–100: 0 = strong no-confidence, 100 = full support
  noConfidenceThreshold: number;     // If voteSentiment drops below this → lose

  // Active issues — unresolved building problems
  activeIssues: BuildingIssue[];

  // Chat history
  chatMessages: ChatMessage[];

  // Flags
  isGameOver: boolean;
  gameResult?: "win" | "lose";
  gameOverReason?: string;
}

type GamePhase =
  | "intro"          // First 30 seconds — calm before storm
  | "tension"        // Normal gameplay
  | "crisis"         // chaosLevel > 70
  | "vote"           // Final 60 seconds — no-confidence vote emerging
  | "ended";

interface BuildingIssue {
  id: string;
  type: IssueType;
  severity: number;           // 0–100
  reportedBy: string;         // Resident ID
  createdAt: number;          // tick
  resolvedAt?: number;        // tick
  isResolved: boolean;
}

type IssueType =
  | "elevator_broken"
  | "water_leak"
  | "noise_complaint"
  | "cleanliness"
  | "parking_dispute"
  | "pet_complaint"
  | "renovation_noise"
  | "maintenance_fee_dispute";

interface ChatMessage {
  id: string;
  senderId: string;           // Resident ID or "player"
  senderName: string;
  content: string;            // Generated by LLM or typed by player
  timestamp: number;          // tick
  type: "resident" | "player" | "system";
  relatedEventId?: string;
}
```

## Thresholds (tunable constants)

```typescript
const THRESHOLDS = {
  NOISE_COMPLAINT_TRIGGER: 60,       // noiseLevel above this → noise events
  CHAOS_CRISIS_TRIGGER: 70,          // chaosLevel above this → crisis phase
  CHAOS_LOSE_TRIGGER: 95,            // chaosLevel above this → lose
  NO_CONFIDENCE_LOSE: 25,            // voteSentiment below this → lose
  STABILITY_DECAY_RATE: 0.5,         // Per tick, stability drifts toward chaos
  ANGER_DECAY_RATE: 2,               // Resident anger drops by this per tick (baseline)
};
```
